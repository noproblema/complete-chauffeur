defined('ABSPATH') || exit;

// Define plugin constants
define('CHAUFFEUR_BOOKING_PATH', plugin_dir_path(__FILE__));
define('CHAUFFEUR_BOOKING_URL', plugin_dir_url(__FILE__));
define('CHAUFFEUR_BOOKING_BASENAME', plugin_basename(__FILE__));
define('CHAUFFEUR_BOOKING_VERSION', '1.0.0');

// Load dependencies
require CHAUFFEUR_BOOKING_PATH . 'includes/class-container.php';
require CHAUFFEUR_BOOKING_PATH . 'includes/class-autoloader.php';

// Initialize the DI Container
$container = new Chauffeur_DIContainer();

// Configure the container
$container_config = require CHAUFFEUR_BOOKING_PATH . 'includes/container-config.php';
$container_config($container);

// Register activation/deactivation hooks
register_activation_hook(__FILE__, function() use ($container) {
    $db_schema = $container->get('Chauffeur_DB_Schema');
    $db_schema->activate();
    
    // Rebuild classmap cache on activation
    Chauffeur_Autoloader::scan_directory();
    Chauffeur_Autoloader::cache_classmap();
});

register_deactivation_hook(__FILE__, function() use ($container) {
    $db_schema = $container->get('Chauffeur_DB_Schema');
    $db_schema->deactivate();
});

// Main plugin class
final class Chauffeur_Booking_Plugin {
    private $container;
    
    public function __construct(Chauffeur_DIContainer $container) {
        $this->container = $container;
        $this->init_hooks();
    }
    
    private function init_hooks() {
        // Initialize the autoloader
        Chauffeur_Autoloader::register();
        
        // Load text domain
        add_action('plugins_loaded', [$this, 'load_textdomain']);
        
        // Initialize components
        add_action('init', [$this, 'init_components']);
    }
    
    public function load_textdomain() {
        load_plugin_textdomain(
            'chauffeur-booking',
            false,
            dirname(CHAUFFEUR_BOOKING_BASENAME) . '/languages/'
        );
    }
    
    public function init_components() {
        // Initialize frontend components
        $this->container->get('Chauffeur_Public_Shortcode');
        $this->container->get('FormHandlerInterface');
        
        // Initialize admin components
        if (is_admin()) {
            $this->container->get('Chauffeur_Admin_Menu');
            $this->container->get('Chauffeur_Admin_Assets');
        }
    }
}

// Initialize the plugin

class Chauffeur_Admin_Menu {
    private $bookings_admin;
    private $routes_admin;
    private $vehicles_admin;

    public function __construct(
        BookingsAdminInterface $bookings_admin,
        RoutesAdminInterface $routes_admin,
        VehiclesAdminInterface $vehicles_admin
    ) {
        $this->bookings_admin = $bookings_admin;
        $this->routes_admin = $routes_admin;
        $this->vehicles_admin = $vehicles_admin;
        
        add_action('admin_menu', [$this, 'add_admin_menus']);
        add_action('admin_init', [$this, 'handle_global_actions']);
    }

    public function add_admin_menus() {
        // Main Bookings Menu
        add_menu_page(
            __('Bookings Calendar', 'chauffeur-booking'),
            __('Chauffeur Bookings', 'chauffeur-booking'),
            'manage_chauffeur_bookings',
            'chauffeur-bookings',
            [$this, 'render_calendar_page'],
            'dashicons-calendar',
            6
        );

        // Submenu Items
        add_submenu_page(
            'chauffeur-bookings',
            __('Bookings Table', 'chauffeur-booking'),
            __('All Bookings', 'chauffeur-booking'),
            'manage_chauffeur_bookings',
            'chauffeur-bookings',
            [$this->bookings_admin, 'render_page']
        );

        add_submenu_page(
            'chauffeur-bookings',
            __('Routes Editor', 'chauffeur-booking'),
            __('Routes', 'chauffeur-booking'),
            'manage_chauffeur_routes',
            'chauffeur-routes',
            [$this->routes_admin, 'render_page']
        );

        add_submenu_page(
            'chauffeur-bookings',
            __('Manage Vehicles', 'chauffeur-booking'),
            __('Vehicles', 'chauffeur-booking'),
            'manage_chauffeur_vehicles',
            'chauffeur-vehicles',
            [$this->vehicles_admin, 'render_page']
        );
    }

    public function render_calendar_page() {
        // Delegate to bookings admin for calendar rendering
        $this->bookings_admin->render_calendar();
    }

    public function handle_global_actions() {
        // Handle bulk actions across admin pages
        if (isset($_REQUEST['action']) && current_user_can('manage_chauffeur_bookings')) {
            $this->handle_bulk_actions();
        }
    }

    private function handle_bulk_actions() {
        $action = $_REQUEST['action'];
        $ids = $_REQUEST['ids'] ?? [];

        if (!wp_verify_nonce($_REQUEST['_wpnonce'], 'bulk-bookings')) {
            wp_die(__('Security check failed', 'chauffeur-booking'));
        }

        try {
            switch ($action) {
                case 'delete':
                    foreach ($ids as $id) {
                        $this->bookings_admin->delete(absint($id));
                    }
                    break;
                case 'export':
                    $this->bookings_admin->export_bookings($ids);
                    break;
            }
        } catch (Exception $e) {
            wp_die($e->getMessage());
        }
    }
}

class Chauffeur_Bookings_Admin implements BookingsAdminInterface {
    private $repository;
    private $notices = [];

    public function __construct(BookingRepositoryInterface $repository) {
        $this->repository = $repository;
        add_action('admin_init', [$this, 'handle_form_submission']);
    }

    // Interface Method: render_page()
    public function render_page(): void {
        $action = $_GET['action'] ?? '';
        $id = $_GET['id'] ?? 0;

        switch ($action) {
            case 'edit':
                $this->render_edit_form(absint($id));
                break;
            case 'add':
                $this->render_add_form();
                break;
            default:
                $this->render_list_table();
        }
    }

    // Interface Method: render_calendar()
    public function render_calendar(): void {
        wp_enqueue_script('tui-calendar');
        wp_enqueue_style('tui-calendar-css');

        $start_date = isset($_GET['date']) ? sanitize_text_field($_GET['date']) : date('Y-m-d');
        $bookings = $this->repository->get_by_date_range(
            $start_date,
            date('Y-m-d', strtotime($start_date . ' +7 days'))
        );

        include CHAUFFEUR_BOOKING_PATH . 'admin/views/bookings/calendar.php';
    }

    // Interface Method: delete()
    public function delete(int $id): bool {
        check_admin_referer('delete_booking_' . $id);
        
        try {
            $result = $this->repository->delete($id);
            if ($result) {
                $this->add_notice(__('Booking deleted successfully!', 'chauffeur-booking'));
                return true;
            }
            throw new RuntimeException(__('Failed to delete booking', 'chauffeur-booking'));
        } catch (Exception $e) {
            $this->add_notice($e->getMessage(), 'error');
            return false;
        }
    }

    // Interface Method: export_bookings()
    public function export_bookings(array $ids): void {
        check_admin_referer('bulk-export');
        
        header('Content-Type: text/csv');
        header('Content-Disposition: attachment; filename="bookings-export-' . date('Y-m-d') . '.csv"');
        
        $output = fopen('php://output', 'w');
        
        // CSV headers
        fputcsv($output, [
            __('ID', 'chauffeur-booking'),
            __('Name', 'chauffeur-booking'),
            __('Pickup Date', 'chauffeur-booking'),
            __('Status', 'chauffeur-booking')
        ]);
        
        // Data rows
        foreach ($ids as $id) {
            $booking = $this->repository->find(absint($id));
            if ($booking) {
                fputcsv($output, [
                    $booking->id,
                    $booking->name,
                    $booking->pickup_date,
                    $booking->status
                ]);
            }
        }
        
        fclose($output);
        exit;
    }

    // Interface Method: handle_form_submission()
    public function handle_form_submission(): void {
        if (!isset($_POST['booking_nonce'])) return;
        
        check_admin_referer('booking_action', 'booking_nonce');

        try {
            $data = $this->sanitize_booking_data($_POST);
            
            if (!empty($_POST['booking_id'])) {
                $this->repository->update(absint($_POST['booking_id']), $data);
                $this->add_notice(__('Booking updated successfully!', 'chauffeur-booking'));
            } else {
                $this->repository->save($data);
                $this->add_notice(__('Booking created successfully!', 'chauffeur-booking'));
            }
        } catch (Exception $e) {
            $this->add_notice($e->getMessage(), 'error');
        }
    }

    // Private helper methods
    private function render_edit_form(int $id): void {
        $booking = $this->repository->find($id);
        if (!$booking) {
            wp_die(__('Booking not found', 'chauffeur-booking'));
        }

        include CHAUFFEUR_BOOKING_PATH . 'admin/views/bookings/edit.php';
    }

    private function render_add_form(): void {
        include CHAUFFEUR_BOOKING_PATH . 'admin/views/bookings/add.php';
    }

    private function render_list_table(): void {
        $bookings_table = new Chauffeur_Bookings_List_Table($this->repository);
        $bookings_table->prepare_items();
        
        include CHAUFFEUR_BOOKING_PATH . 'admin/views/bookings/list.php';
    }

    private function sanitize_booking_data(array $data): array {
        return [
            'name' => sanitize_text_field($data['name'] ?? ''),
            'email' => sanitize_email($data['email'] ?? ''),
            'phone' => sanitize_text_field($data['phone'] ?? ''),
            'vehicle' => sanitize_text_field($data['vehicle'] ?? ''),
            'route_id' => absint($data['route_id'] ?? 0),
            'pickup_date' => sanitize_text_field($data['pickup_date'] ?? ''),
            'pickup_location' => sanitize_text_field($data['pickup_location'] ?? ''),
            'destination' => sanitize_text_field($data['destination'] ?? ''),
            'distance' => floatval($data['distance'] ?? 0),
            'price' => floatval($data['price'] ?? 0),
            'status' => $this->sanitize_status($data['status'] ?? 'pending')
        ];
    }

    private function sanitize_status(string $status): string {
        $valid_statuses = ['pending', 'confirmed', 'completed', 'cancelled'];
        return in_array($status, $valid_statuses) ? $status : 'pending';
    }

    private function add_notice(string $message, string $type = 'success'): void {
        $this->notices[] = [
            'message' => $message,
            'type' => $type
        ];
        
        // Also store in transients for page reloads
        set_transient(
            'chauffeur_admin_notice_' . get_current_user_id(),
            ['message' => $message, 'type' => $type],
            45
        );
    }

    private function display_notices(): void {
        // Show immediate notices
        foreach ($this->notices as $notice) {
            printf(
                '<div class="notice notice-%s is-dismissible"><p>%s</p></div>',
                esc_attr($notice['type']),
                esc_html($notice['message'])
            );
        }
        
        // Show persistent notices (from transients)
        if ($notice = get_transient('chauffeur_admin_notice_' . get_current_user_id())) {
            printf(
                '<div class="notice notice-%s is-dismissible"><p>%s</p></div>',
                esc_attr($notice['type']),
                esc_html($notice['message'])
            );
            delete_transient('chauffeur_admin_notice_' . get_current_user_id());
        }
    }
}

jQuery(document).ready(function ($) {
    // Define the ChauffeurCalendar class
    class ChauffeurCalendar extends tui.Calendar {
        constructor(container, options) {
            // Call the parent class constructor
            super(container, options);

            // Initialize the calendar
            this.initialize();
        }

        /**
         * Initialize the calendar with event listeners and default settings.
         */
        initialize() {
            // Add event listener for date/time selection
            this.on('selectDateTime', (event) => this.handleDateTimeSelect(event));

            // Fetch bookings and add them to the calendar
            this.fetchBookings();

            // Add navigation event handlers
            this.addNavigationHandlers();

            // Add view switch event handlers (month/day)
            this.addViewSwitchHandlers();

            // Initialize the Toast UI DatePicker
            this.initializeDatePicker();
        }

        /**
         * Handle date/time selection.
         * @param {Object} event - The event object containing the selected date/time.
         */
        handleDateTimeSelect(event) {
            const clickedDate = event.start; // Get the clicked date
            this.setDate(clickedDate); // Set calendar date to the clicked date
            this.updateCurrentSelection(); // Update the current selection display
            this.updateDatePicker(); // Update the datepicker
        }

        /**
         * Fetch bookings and add them to the calendar.
         */
        fetchBookings() {
            $.ajax({
                url: ajaxurl,
                type: 'POST',
                data: {
                    action: 'fetch_bookings',
                    nonce: chauffeur_admin_ajax.nonce
                },
                success: (response) => {
                    if (response.success) {
                        // Add bookings to the calendar
                        this.createEvents(response.data);
                    } else {
                        console.error('Failed to fetch bookings:', response.data);
                    }
                },
                error: (xhr, status, error) => {
                    console.error('AJAX error:', error);
                }
            });
        }

        /**
         * Add navigation event handlers for Previous, Next, and Today buttons.
         */
        addNavigationHandlers() {
            // Previous button
            $('#prev-button').on('click', () => {
                this.prev();
                this.updateCurrentSelection();
                this.updateDatePicker(); // Update the datepicker
            });

            // Next button
            $('#next-button').on('click', () => {
                this.next();
                this.updateCurrentSelection();
                this.updateDatePicker(); // Update the datepicker
            });

            // Today button
            $('#today-button').on('click', () => {
                this.today();
                this.updateCurrentSelection();
                this.updateDatePicker(); // Update the datepicker
            });
        }

        /**
         * Add event handlers for switching between month and day views.
         */
        addViewSwitchHandlers() {
            // Month view button
            $('#month-view').on('click', () => {
                this.changeView('month');
                this.updateCurrentSelection();
                this.updateButtonActiveState('month');
                this.updateDatePicker(); // Update the datepicker
            });

            // Day view button
            $('#day-view').on('click', () => {
                this.changeView('day');
                this.updateCurrentSelection();
                this.updateButtonActiveState('day');
                this.updateDatePicker(); // Update the datepicker
            });
        }

        /**
         * Initialize the Toast UI DatePicker.
         */
        initializeDatePicker() {
            const $datepickerInput = $('#datepicker-input');
            const $datepickerContainer = $('#datepicker-container');

            // Initialize the Toast UI DatePicker
            this.datepicker = new tui.DatePicker($datepickerContainer[0], {
                date: this.getDate().toDate(), // Set initial date
                input: {
                    element: $datepickerInput[0], // Input element for the datepicker
                    format: 'd/M/yyyy' // Format: dd/MM/yyyy
                },
                language: 'en', // Set language
                showAlways: false, // Do not show the datepicker by default
                selectableRanges: [
                    // Define selectable date range (optional)
                    [new Date(2000, 0, 1), new Date(2100, 11, 31)]
                ]
            });

            // Handle date selection
            this.datepicker.on('change', () => {
                const selectedDate = this.datepicker.getDate(); // Get the selected date from the datepicker object
                console.log(`Selected date: ${selectedDate}`); // Log the selected date (for debugging)
                this.setDate(selectedDate); // Set the calendar date
                this.updateCurrentSelection(); // Update the current selection display
            });
        }

        /**
         * Update the current selection display.
         */
        updateCurrentSelection() {
            const currentDate = this.getDate().toDate(); // Convert tzDate to native Date
            const viewName = this.getViewName();

            let selectionText = '';

            if (viewName === 'month') {
                // Format: Month Year (e.g., January 2023)
                selectionText = currentDate.toLocaleString('default', { month: 'long', year: 'numeric' });
            } else if (viewName === 'day') {
                // Format: Weekday, d MMM yyyy (e.g., Monday, 2 Jan 2023)
                selectionText = currentDate.toLocaleString('default', { weekday: 'long', day: 'numeric', month: 'short', year: 'numeric' });
            }

            $('#current-selection').text(selectionText);
        }

        /**
         * Update the Toast UI DatePicker to reflect the current calendar date.
         */
        updateDatePicker() {
            const currentDate = this.getDate().toDate(); // Convert tzDate to native Date

            if (this.datepicker) {
                // Set the datepicker value
                this.datepicker.setDate(currentDate);
            }
        }

        /**
         * Update the active state of the view buttons (month/day).
         * @param {string} activeView - The currently active view ('month' or 'day').
         */
        updateButtonActiveState(activeView) {
            $('#month-view, #day-view').removeClass('active');
            $(`#${activeView}-view`).addClass('active');
        }
    }

    // Check if the current page is the calendar page
    if ($('body').hasClass('toplevel_page_chauffeur-bookings')) {
        // Initialize the ChauffeurCalendar
        const calendar = new ChauffeurCalendar('#calendar', {
            defaultView: 'month', // Default to month view
            useCreationPopup: true,
            useDetailPopup: true,

class Chauffeur_DIContainer {
    private $definitions = [];
    private $instances = [];

    public function __construct(array $definitions = []) {
        $this->definitions = $definitions;
    }

    public function set(string $id, $concrete): void {
        $this->definitions[$id] = $concrete;
        unset($this->instances[$id]); // Remove cached instance if exists
    }

    public function get(string $id) {
        // Return cached instance if available
        if (isset($this->instances[$id])) {
            return $this->instances[$id];
        }

        // Create new instance
        $concrete = $this->definitions[$id] ?? $id;
        
        if (is_callable($concrete)) {
            $instance = $concrete($this);
        } elseif (is_string($concrete) && class_exists($concrete)) {
            $instance = $this->autowire($concrete);
        } else {
            $instance = $concrete;
        }

        // Cache the instance
        $this->instances[$id] = $instance;
        return $instance;
    }

    private function autowire(string $class) {
        $reflection = new ReflectionClass($class);
        
        // Check if class is instantiable
        if (!$reflection->isInstantiable()) {
            throw new RuntimeException("Class {$class} is not instantiable");
        }

        // Get constructor params
        $constructor = $reflection->getConstructor();
        if (is_null($constructor)) {
            return new $class();
        }

        // Resolve dependencies
        $parameters = $constructor->getParameters();
        $dependencies = array_map(
            function ($param) {
                return $this->get($this->getParameterClassName($param));
            },
            $parameters
        );

        return $reflection->newInstanceArgs($dependencies);
    }

    private function getParameterClassName(ReflectionParameter $param): string {
        $type = $param->getType();
        
        if (!$type || $type->isBuiltin()) {
            throw new RuntimeException(
                "Cannot resolve parameter {$param->name} in {$param->getDeclaringClass()->name}"
            );
        }

        return $type->getName();
    }
}

class Chauffeur_Form_Handler implements FormHandlerInterface {
    private $booking_service;
    private $route_repository;
    private $vehicle_repository;
    private $submitted_data = [];
    private $transient_key = 'chauffeur_booking_notice';

    public function __construct(
        BookingServiceInterface $booking_service,
        RouteRepositoryInterface $route_repository,
        VehicleRepositoryInterface $vehicle_repository
    ) {
        $this->booking_service = $booking_service;
        $this->route_repository = $route_repository;
        $this->vehicle_repository = $vehicle_repository;
        $this->init_hooks();
    }

    public function get_routes(): array {
        return $this->route_repository->get_all();
    }

    public function get_vehicles(): array {
        return $this->vehicle_repository->get_all();
    }

    public function get_field_value(string $field, $default = ''): string {
        // Get from submitted data if available
        if (isset($this->submitted_data[$field])) {
            return $this->submitted_data[$field];
        }
        
        // Fallback to default or empty string
        return $default;
    }

    private function init_hooks(): void {
        add_action('init', [$this, 'maybe_handle_submission']);
        add_action('wp_enqueue_scripts', [$this, 'enqueue_assets']);
        add_action('wp_ajax_chauffeur_submit_booking', [$this, 'handle_ajax_submission']);
        add_action('wp_ajax_nopriv_chauffeur_submit_booking', [$this, 'handle_ajax_submission']);
    }

    public function enqueue_assets(): void {
        wp_enqueue_style(
            'chauffeur-booking-form',
            CHAUFFEUR_BOOKING_URL . 'assets/css/frontend.css',
            [],
            CHAUFFEUR_BOOKING_VERSION
        );

        wp_enqueue_script(
            'chauffeur-booking-form',
            CHAUFFEUR_BOOKING_URL . 'assets/js/frontend.js',
            ['jquery', 'flatpickr'],
            CHAUFFEUR_BOOKING_VERSION,
            true
        );

        wp_localize_script('chauffeur-booking-form', 'chauffeurBooking', [
            'ajax_url' => admin_url('admin-ajax.php'),
            'nonce' => wp_create_nonce('chauffeur_booking_nonce')
        ]);
    }

    public function render_form(): string {
        ob_start();
        include CHAUFFEUR_BOOKING_PATH . 'templates/booking-form.php';
        return ob_get_clean();
    }

    public function maybe_handle_submission(): void {
        if ($this->is_form_submitted()) {
            $this->handle_submission();
        }
    }

    private function is_form_submitted(): bool {
        return isset($_POST['chbo_submit']) && 
               wp_verify_nonce($_POST['chauffeur_booking_nonce'], 'chauffeur_booking_form');
    }

    public function handle_ajax_submission(): void {
        check_ajax_referer('chauffeur_booking_nonce', 'nonce');

        try {
            $data = $this->sanitize_input($_POST);
            $booking_id = $this->booking_service->create_booking($data);
            
            wp_send_json_success([
                'message' => __('Booking successful!', 'chauffeur-booking'),
                'booking_id' => $booking_id
            ]);
        } catch (Exception $e) {
            wp_send_json_error([
                'message' => $e->getMessage()
            ]);
        }
    }

    private function handle_submission(): void {
        $this->submitted_data = $this->sanitize_input($_POST);

        try {
            $booking_id = $this->booking_service->create_booking($this->submitted_data);
            $this->set_notice(__('Booking created successfully!', 'chauffeur-booking'));
        } catch (Exception $e) {
            $this->set_notice($e->getMessage(), 'error');
        }
    }

    private function sanitize_input(array $input): array {
        return [
            'name' => sanitize_text_field($input['chbo_name'] ?? ''),
            'email' => sanitize_email($input['chbo_email'] ?? ''),
            'phone' => sanitize_text_field($input['chbo_phone'] ?? ''),
            'vehicle' => sanitize_text_field($input['chbo_vehicle'] ?? ''),
            'route_id' => absint($input['chbo_route'] ?? 0),
            'pickup_date' => sanitize_text_field($input['chbo_pickup_date'] ?? ''),
            'pickup_location' => sanitize_text_field($input['chbo_pickup_location'] ?? ''),
            'destination' => sanitize_text_field($input['chbo_destination'] ?? ''),
            'distance' => floatval($input['chbo_distance'] ?? 0),
            'price' => floatval($input['chbo_price'] ?? 0),
            'roundtrip' => isset($input['chbo_roundtrip']) ? 1 : 0,
            'notes' => sanitize_textarea_field($input['chbo_notes'] ?? '')
        ];
    }

    private function set_notice(string $message, string $type = 'success'): void {
        set_transient($this->transient_key, [
            'message' => $message,
            'type' => $type
        ], 30);
    }

    public function display_notices(): void {
        if ($notice = get_transient($this->transient_key)) {
            printf(
                '<div class="notice notice-%s"><p>%s</p></div>',
                esc_attr($notice['type']),
                esc_html($notice['message'])
            );
            delete_transient($this->transient_key);
        }
    }
}

        }

        $base_price = !empty($booking_data['roundtrip']) 
            ? $route->roundtrip_price 
            : $route->price;

        return $base_price * ($vehicle->price_rate / 100);

interface BookingRepositoryInterface {
    public function find(int $id): ?object;
    public function get_all(array $args = []): array;
    public function get_paginated(int $per_page = 10, int $page = 1): array;
    public function get_by_date_range(string $start_date, string $end_date): array;
    public function save(array $data): int;
    public function update(int $id, array $data): bool;
    public function delete(int $id): bool;
    public function update_status(int $id, string $status): bool;
}
